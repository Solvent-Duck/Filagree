CREATE OR REPLACE PROCEDURE FILAGREE.MART.REFRESH_MARTS()
RETURNS STRING
LANGUAGE SQL
AS
$$
BEGIN
  -- Net revenue per region, nation, month and market segment.
  CREATE OR REPLACE TABLE FILAGREE.MART.REVENUE_BY_NATION_MONTH AS
  SELECT
    r.r_name                                    AS region,
    n.n_name                                    AS nation,
    DATE_TRUNC('month', o.o_orderdate)          AS month,
    c.c_mktsegment                              AS segment,
    COUNT(DISTINCT o.o_orderkey)                AS orders,
    SUM(l.l_extendedprice * (1 - l.l_discount)) AS net_revenue
  FROM SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.LINEITEM l
  JOIN SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.ORDERS   o ON o.o_orderkey  = l.l_orderkey
  JOIN SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.CUSTOMER c ON c.c_custkey   = o.o_custkey
  JOIN SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.NATION   n ON n.n_nationkey = c.c_nationkey
  JOIN SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.REGION   r ON r.r_regionkey = n.n_regionkey
  GROUP BY ALL;

  -- One row per supplier: what they ship, how late, and how much it earns.
  CREATE OR REPLACE TABLE FILAGREE.MART.SUPPLIER_SCORECARD AS
  SELECT
    s.s_suppkey                                                  AS supplier_key,
    s.s_name                                                     AS supplier,
    n.n_name                                                     AS nation,
    COUNT(DISTINCT p.p_partkey)                                  AS parts_shipped,
    SUM(l.l_extendedprice * (1 - l.l_discount))                  AS net_revenue,
    SUM(l.l_extendedprice * (1 - l.l_discount))
      - SUM(ps.ps_supplycost * l.l_quantity)                     AS margin,
    AVG(IFF(l.l_receiptdate > l.l_commitdate, 1, 0))             AS late_share
  FROM SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.LINEITEM l
  JOIN SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.SUPPLIER s  ON s.s_suppkey   = l.l_suppkey
  JOIN SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.PARTSUPP ps ON ps.ps_partkey = l.l_partkey
                                                 AND ps.ps_suppkey = l.l_suppkey
  JOIN SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.PART     p  ON p.p_partkey   = l.l_partkey
  JOIN SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.NATION   n  ON n.n_nationkey = s.s_nationkey
  GROUP BY ALL;

  RETURN 'marts rebuilt';
END;

-- Plain-English docs for every source table and mart, written from the column catalog.
CREATE OR REPLACE TABLE FILAGREE.AI.TABLE_DOCS AS
WITH catalog AS (
  SELECT table_catalog, table_schema, table_name,
         LISTAGG(column_name || ' ' || data_type, ', ')
           WITHIN GROUP (ORDER BY ordinal_position) AS columns
  FROM SNOWFLAKE_SAMPLE_DATA.INFORMATION_SCHEMA.COLUMNS
  WHERE table_schema = 'TPCH_SF1'
  GROUP BY ALL
  UNION ALL
  SELECT table_catalog, table_schema, table_name,
         LISTAGG(column_name || ' ' || data_type, ', ')
           WITHIN GROUP (ORDER BY ordinal_position)
  FROM FILAGREE.INFORMATION_SCHEMA.COLUMNS
  WHERE table_schema = 'MART'
  GROUP BY ALL
)
SELECT
  table_catalog || '.' || table_schema || '.' || table_name AS table_fqn,
  table_name,
  columns,
  AI_COMPLETE(
    'llama3.3-70b',
    'You document data for people who will never read SQL. In two short sentences, '
    || 'say what one row of the table ' || table_name || ' represents and one question '
    || 'it can answer. Columns: ' || columns
  )                    AS summary,
  'llama3.3-70b'       AS model,
  CURRENT_TIMESTAMP()  AS generated_at

-- Answer a spoken question. focus_table is the box the viewer has selected (by gesture), or NULL.
CREATE OR REPLACE PROCEDURE FILAGREE.AI.ASK_LAKE(question STRING, focus_table STRING DEFAULT NULL)
RETURNS STRING
LANGUAGE SQL
AS
$$
DECLARE
  docs STRING;
  revenue STRING;
  suppliers STRING;
  answer STRING;
BEGIN
  SELECT LISTAGG(table_name || ': ' || summary, '\n') INTO :docs
  FROM FILAGREE.AI.TABLE_DOCS
  WHERE :focus_table IS NULL OR table_name = UPPER(:focus_table)
     OR table_name IN ('REVENUE_BY_NATION_MONTH', 'SUPPLIER_SCORECARD');

  SELECT LISTAGG(region || ' ' || yr || ': ' || TO_VARCHAR(rev, '999,999,999,999'), '\n') INTO :revenue
  FROM (SELECT region, YEAR(month) AS yr, ROUND(SUM(net_revenue)) AS rev
        FROM FILAGREE.MART.REVENUE_BY_NATION_MONTH GROUP BY ALL ORDER BY region, yr);

  SELECT LISTAGG(supplier || ' (' || nation || ') margin ' || ROUND(margin)
                 || ', late ' || ROUND(late_share * 100) || '%', '\n') INTO :suppliers
  FROM (SELECT * FROM FILAGREE.MART.SUPPLIER_SCORECARD ORDER BY margin DESC LIMIT 10);

  SELECT AI_COMPLETE(
    'llama3.3-70b',
    'You answer questions about the TPC-H sample dataset for someone looking at a 3D map '
    || 'of it, not at SQL. Answer in at most three spoken sentences, using only the facts below. '
    || 'Name the table the answer comes from.\n\nTables:\n' || :docs
    || '\n\nFrom table REVENUE_BY_NATION_MONTH, net revenue by region and year:\n' || :revenue
    || '\n\nFrom table SUPPLIER_SCORECARD, top 10 suppliers by margin:\n' || :suppliers
    || '\n\nSelected box: ' || COALESCE(:focus_table, 'none')
    || '\n\nQuestion: ' || :question
  ) INTO :answer;

  RETURN answer;
END;
$$;
